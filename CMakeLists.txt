cmake_minimum_required(VERSION 3.21)
project(nucleus_btc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nucleus_core SHARED native/src/core.cpp)
target_include_directories(nucleus_core PRIVATE native/include)
add_executable(nucleus-native native/src/main.cpp)
target_include_directories(nucleus-native PRIVATE native/include)
if(MSVC)
  target_compile_options(nucleus_core PRIVATE /O2 /W4 /EHsc)
  target_compile_options(nucleus-native PRIVATE /O2 /W4 /EHsc)
else()
  target_compile_options(nucleus_core PRIVATE -O3 -Wall -Wextra)
  target_compile_options(nucleus-native PRIVATE -O3 -Wall -Wextra)
endif()
option(NUCLEUS_ENABLE_HIP "Compile optional exact HIP backend (requires SDK)" OFF)
if(NUCLEUS_ENABLE_HIP)
  enable_language(HIP)
  add_library(nucleus_hip SHARED native/hip/baseline.hip)
  target_include_directories(nucleus_hip PRIVATE native/include)
  set_target_properties(nucleus_hip PROPERTIES HIP_STANDARD 20)
endif()
